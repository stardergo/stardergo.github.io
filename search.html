let posts = []; let timer; const input = document.getElementById('searchInput'); const results = document.getElementById('results'); const count = document.getElementById('resultCount'); // --- 1. 加载 search.json --- async function loadData() { try { const resp = await fetch('/search.json?t=' + Date.now()); if (!resp.ok) throw new Error('文件不存在'); posts = await resp.json(); console.log('✅ 已加载 ' + posts.length + ' 篇文章'); } catch (e) { results.innerHTML = `
  • 📂

    搜索数据加载失败

    请确认 search.json 文件存在

    • `; console.error('加载 search.json 失败:', e); } } // --- 2. 执行搜索（使用已加载的 posts 数据）--- function performSearch(keyword) { const q = keyword.trim().toLowerCase(); if (q.length < 1) { results.innerHTML = ''; count.textContent = ''; return; } const matched = posts.filter(post => { const titleLower = post.title.toLowerCase(); const summaryLower = (post.summary || '').toLowerCase(); return titleLower.includes(q) || summaryLower.includes(q); }); matched.sort((a, b) => { const aTitle = a.title.toLowerCase().includes(q); const bTitle = b.title.toLowerCase().includes(q); if (aTitle && !bTitle) return -1; if (!aTitle && bTitle) return 1; return a.title.localeCompare(b.title); }); displayResults(matched, q); } // --- 3. 显示结果 --- function displayResults(data, q) { if (data.length === 0) { results.innerHTML = `
  • 😅

    没找到包含 “${q}” 的文章

    • `; count.textContent = ''; return; } count.textContent = `找到 ${data.length} 篇文章`; const escaped = q.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); const regex = new RegExp(escaped, 'gi'); results.innerHTML = data.map(item => { const highlightedTitle = item.title.replace(regex, match => `${match}`); const highlightedSummary = (item.summary || '').replace(regex, match => `${match}`); return `
  • ${highlightedTitle}
    ${item.summary ? `
    ${highlightedSummary}
    ` : ''}
    ${item.url}
    • `; }).join(''); } // --- 4. 防抖监听 --- input.addEventListener('input', function() { clearTimeout(timer); const q = this.value.trim(); if (q.length < 1) { results.innerHTML = ''; count.textContent = ''; return; } results.innerHTML = `

  • 搜索中...

    • `; timer = setTimeout(() => performSearch(q), 250); }); // --- 5. 快捷键 --- document.addEventListener('keydown', e => { if ((e.ctrlKey || e.metaKey) && e.key === 'k') { e.preventDefault(); input.focus(); } if (e.key === 'Escape' && document.activeElement === input) { input.value = ''; results.innerHTML = ''; count.textContent = ''; input.blur(); } }); // --- 6. 启动 --- loadData(); window.addEventListener('load', () => setTimeout(() => input.focus(), 300));
